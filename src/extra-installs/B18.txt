RUN (git clone -q https://github.com/theartofcomputerprogramming/mmixware.git /opt/mmixware || git clone -q https://gitlab.lrz.de/mmix/mmixware.git /opt/mmixware) && make -C /opt/mmixware CFLAGS="-O2 -std=gnu89 -Wno-implicit-int -Wno-implicit-function-declaration -Wno-return-type" mmixal mmix && install -m 755 /opt/mmixware/mmixal /opt/mmixware/mmix /usr/local/bin/ && rm -rf /opt/mmixware && printf ' LOC #100\nMain GETA $255,S\n TRAP 0,Fputs,StdOut\n TRAP 0,Halt,0\nS BYTE "ok",0\n' > /tmp/t.mms && mmixal -o /tmp/t.mmo /tmp/t.mms && test "$(mmix /tmp/t.mmo)" = "ok" && rm -f /tmp/t.mms /tmp/t.mmo
RUN mkdir -p /opt/uxn && (curl --retry 5 -fsSL https://github.com/kivutar/uxn/archive/9e483ae8b129029152a2b3e1ebb6762078378e65.tar.gz | tar -xz -C /opt/uxn --strip-components=1 || git clone -q --depth 1 https://github.com/kivutar/uxn.git /opt/uxn) && cd /opt/uxn && gcc -O2 -Wno-error src/uxnasm.c -o /usr/local/bin/uxnasm && (gcc -O2 -Wno-error src/uxn.c src/uxncli.c -o /usr/local/bin/uxncli 2>/dev/null || gcc -O2 -Wno-error src/uxncli.c -o /usr/local/bin/uxncli 2>/dev/null || gcc -O2 -Wno-error src/uxn.c src/devices/*.c src/uxncli.c -o /usr/local/bin/uxncli) && cd / && rm -rf /opt/uxn && printf '|0100 #6f #18 DEO #6b #18 DEO #800f DEO BRK\n' > /tmp/t.tal && uxnasm /tmp/t.tal /tmp/t.rom >/dev/null && test "$(uxncli /tmp/t.rom)" = "ok" && rm -f /tmp/t.tal /tmp/t.rom
