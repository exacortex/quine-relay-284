# B01: Dockerfile lines for installs with no apt package. Pin versions. (Maxima and Gri are apt: Apt = in batch.rb.)
# Build deps for UCBLogo and GNU Smalltalk, unzip for Dafny.
RUN chronic apt-get -qq install -y autoconf automake autoconf-archive libncurses-dev libgmp-dev libffi-dev libltdl-dev libreadline-dev zlib1g-dev gawk pkg-config unzip && chronic apt-get clean
# UCBLogo 6.2.5 built without wx, so it prints to stdout and needs no display (Ubuntu's ucblogo package is the wx GUI build).
# glibc 2.43 has no <termio.h> / struct termio any more (and its <sgtty.h> has no TIOCGETP), so term.c is moved to the
# POSIX termios calls (tcgetattr/tcsetattr; only used when stdin is a terminal) and gets a one-line termio.h.
RUN curl --retry 3 -fsSL -o /tmp/ucb.tgz https://github.com/jrincayc/ucblogo-code/archive/refs/tags/version_6.2.5.tar.gz && echo "668734c775d4f8eddb999094941572c55aaa1951b6936df2ba1f376427bedb3c  /tmp/ucb.tgz" | sha256sum -c && tar xzf /tmp/ucb.tgz -C /tmp && cd /tmp/ucblogo-code-version_6.2.5 && mkdir shim && echo '#include <termios.h>' > shim/termio.h && sed -i 's/struct termio /struct termios /; s/ioctl(0,TCGETA,(char \*)(&tty_cooked))/tcgetattr(0,\&tty_cooked)/; s/ioctl(0,TCSETA,(char \*)(&tty_cbreak))/tcsetattr(0,TCSANOW,\&tty_cbreak)/; s/ioctl(0,TCSETA,(char \*)(&tty_cooked))/tcsetattr(0,TCSANOW,\&tty_cooked)/' term.c && ! grep -q 'TCGETA\|TCSETA' term.c && chronic autoreconf -fi && chronic ./configure --disable-docs CFLAGS="-O2 -std=gnu17" CPPFLAGS="-I/tmp/ucblogo-code-version_6.2.5/shim" ac_cv_header_sgtty_h=no && chronic make -j8 && chronic make install && cd / && rm -rf /tmp/ucblogo-code-version_6.2.5 /tmp/ucb.tgz
# GNU Smalltalk 3.2.5 (the last release; Ubuntu dropped it in 23.10). Old C, so gnu11 and gcc 14+'s new errors turned back into warnings.
RUN curl --retry 5 -fsSL -o /tmp/st.tgz https://mirrors.kernel.org/gnu/smalltalk/smalltalk-3.2.5.tar.gz && echo "06e574e818ec49f0555d948ae53b1453d8c1df59ef597dad911a4fd1ffba1cce  /tmp/st.tgz" | sha256sum -c && tar xzf /tmp/st.tgz -C /tmp && cd /tmp/smalltalk-3.2.5 && chronic ./configure --disable-generational-gc --without-emacs --without-tk --without-x CFLAGS="-O2 -std=gnu11 -fcommon -Wno-error -Wno-implicit-function-declaration -Wno-incompatible-pointer-types -Wno-int-conversion" && chronic make -j8 && chronic make install && cd / && rm -rf /tmp/smalltalk-3.2.5 /tmp/st.tgz
# Julia 1.13.4, official binary tarball (Ubuntu dropped the julia package).
RUN curl --retry 3 -fsSL https://julialang-s3.julialang.org/bin/linux/x64/1.13/julia-1.13.4-linux-x86_64.tar.gz | tar xz -C /opt && ln -s /opt/julia-1.13.4/bin/julia /usr/local/bin/julia
# Dafny 4.11.0 release zip (runs on the image's .NET; Z3 is bundled).
RUN curl --retry 3 -fsSL -o /tmp/dafny.zip https://github.com/dafny-lang/dafny/releases/download/v4.11.0/dafny-4.11.0-x64-ubuntu-22.04.zip && echo "a46a9ff7cdd720f7955854c78e95df13f4cfe6b80691b05f8654fe19e8267179  /tmp/dafny.zip" | sha256sum -c && unzip -q /tmp/dafny.zip -d /opt && rm /tmp/dafny.zip && printf '#!/bin/sh\nexec /opt/dafny/dafny "$@"\n' > /usr/local/bin/dafny && chmod +x /usr/local/bin/dafny
# dafny build restores the net8 reference packs from NuGet the first time: do it once here so the lap needs no network.
RUN cd /tmp && echo 'method Main(){print "ok";}' > w.dfy && dafny build -o w/w w.dfy && DOTNET_ROLL_FORWARD=Major dotnet w/w.dll && rm -rf /tmp/w /tmp/w.dfy
