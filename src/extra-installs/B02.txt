# B02: old Ubuntu .debs from archive.ubuntu.com (pinned pool URLs). Ubuntu dropped these packages; the binaries still run on 26.04.
# A+ 4.22.1 and Lisaac 0.39~rc1 from noble (24.04): plain apt install, their deps are all in 26.04
RUN cd /tmp && U=http://archive.ubuntu.com/ubuntu/pool/universe && curl -fsSO $U/a/aplus-fsf/aplus-fsf_4.22.1-12_amd64.deb && curl -fsSO "$U/l/lisaac/lisaac_0.39~rc1-3.1_amd64.deb" && curl -fsSO "$U/l/lisaac/lisaac-common_0.39~rc1-3.1_all.deb" && apt-get update -qq && apt-get install -y -qq ./aplus-fsf_*.deb ./lisaac_*.deb ./lisaac-common_*.deb libcurl3t64-gnutls libexpat1 && apt-get clean && rm -f /tmp/*.deb
# Falcon 0.9.6.9 from trusty (14.04) with its libpcre3 + libssl1.0.0 kept private in /opt/falcon
RUN mkdir -p /opt/falcon && cd /tmp && U=http://archive.ubuntu.com/ubuntu/pool && curl -fsSO $U/universe/f/falconpl/falconpl_0.9.6.9-git20120606-2_amd64.deb && curl -fsSO $U/universe/f/falconpl/libfalcon-engine1_0.9.6.9-git20120606-2_amd64.deb && curl -fsSO $U/main/o/openssl/libssl1.0.0_1.0.1f-1ubuntu2_amd64.deb && curl -fsSO $U/main/p/pcre3/libpcre3_8.31-2ubuntu2_amd64.deb && for d in *.deb; do dpkg -x $d /opt/falcon; done && rm -f /tmp/*.deb && printf '#!/bin/sh\nLD_LIBRARY_PATH=/opt/falcon/usr/lib:/opt/falcon/lib/x86_64-linux-gnu exec /opt/falcon/usr/bin/falcon "$@"\n' > /usr/local/bin/falcon && chmod +x /usr/local/bin/falcon
# Boo 0.9.5 from trusty: CLI assemblies, run on the image's mono
RUN mkdir -p /opt/boo && cd /tmp && U=http://archive.ubuntu.com/ubuntu/pool/universe/b/boo && curl -fsSO "$U/boo_0.9.5~git20110729.r1.202a430-2_all.deb" && curl -fsSO "$U/libboo2.0.9-cil_0.9.5~git20110729.r1.202a430-2_all.deb" && for d in *.deb; do dpkg -x $d /opt/boo; done && rm -f /tmp/*.deb && cp /opt/boo/usr/lib/cli/*/*.dll /opt/boo/usr/lib/boo/ && printf '#!/bin/sh\nexec mono /opt/boo/usr/lib/boo/booi.exe "$@"\n' > /usr/local/bin/booi && chmod +x /usr/local/bin/booi
# eC (ecere-sdk 0.44.15) from focal (20.04): ecp/ecc/ecs + libecereCOM; its GUI libs are already in the image
RUN mkdir -p /opt/ecere && cd /tmp && U=http://archive.ubuntu.com/ubuntu/pool/universe/e/ecere-sdk && for p in ecere-dev libecc0 libecerecom0 libecere0; do curl -fsSO $U/${p}_0.44.15-1build3_amd64.deb; done && for d in *.deb; do dpkg -x $d /opt/ecere; done && rm -f /tmp/*.deb && for t in ecp ecc ecs; do printf '#!/bin/sh\nLD_LIBRARY_PATH=/opt/ecere/usr/lib/x86_64-linux-gnu:/opt/ecere/usr/lib/x86_64-linux-gnu/ec exec /opt/ecere/usr/bin/%s "$@"\n' $t > /usr/local/bin/$t && chmod +x /usr/local/bin/$t; done && ln -s /opt/ecere/usr/lib/x86_64-linux-gnu/libecereCOM.so /usr/local/lib/libecereCOM.so && echo /opt/ecere/usr/lib/x86_64-linux-gnu > /etc/ld.so.conf.d/ecere.conf && ldconfig
