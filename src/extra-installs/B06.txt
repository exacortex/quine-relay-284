# B06: Dockerfile lines for installs with no apt package. Pin versions.
# Merge patch (milestone 248): cargo's parallel HTTP/2 downloads kept dropping TLS sessions on this PC (GnuTLS -10, laps 1-2).
ENV CARGO_HTTP_MULTIPLEXING=false CARGO_NET_RETRY=10
RUN mkdir -p /opt/pwsh && curl -fsSL https://github.com/PowerShell/PowerShell/releases/download/v7.4.6/powershell-7.4.6-linux-x64.tar.gz | tar -xz -C /opt/pwsh && chmod +x /opt/pwsh/pwsh && ln -sf /opt/pwsh/pwsh /usr/local/bin/pwsh
RUN mkdir -p /opt/nu && curl -fsSL https://github.com/nushell/nushell/releases/download/0.101.0/nu-0.101.0-x86_64-unknown-linux-gnu.tar.gz | tar -xz -C /opt/nu --strip-components=1 && ln -sf /opt/nu/nu /usr/local/bin/nu
RUN cargo install --git https://github.com/redox-os/ion.git --rev e9b3ca48b7ab6444498d5c3aa902d471ce29f759 ion-shell --root /usr/local
RUN mkdir -p /tmp/oils && curl -fsSL https://www.oilshell.org/download/oils-for-unix-0.24.0.tar.gz | tar -xz -C /tmp/oils --strip-components=1 && (cd /tmp/oils && ./configure && _build/oils.sh && ./install) && rm -rf /tmp/oils
RUN git clone --branch v7.4.2002 --depth 1 https://github.com/lmorg/murex.git /tmp/murex && (cd /tmp/murex && go build -o /usr/local/bin/murex .) && rm -rf /tmp/murex
RUN cargo install hush --version 0.1.4 --root /usr/local
RUN mkdir -p /tmp/es && curl -fsSL https://github.com/wryun/es-shell/releases/download/v0.9.2/es-0.9.2.tar.gz | tar -xz -C /tmp/es && (cd /tmp/es && ./configure CFLAGS="-O2 -std=gnu89" && make -j4 && cp es /usr/local/bin/es) && rm -rf /tmp/es
