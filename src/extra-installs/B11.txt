# B11: array languages. All pinned.
# Uiua 0.19.1 from crates.io (Rust build, about 6.5 min).
RUN cargo install uiua --version 0.19.1 --locked --root /opt/uiua > /tmp/uiua-build.log 2>&1 && ln -s /opt/uiua/bin/uiua /usr/local/bin/uiua && uiua --version
# CBQN v0.9.0 (BQN) source build. Merge fix: B07 Odin links /usr/local/bin/clang -> gcc, and gcc rejects CBQN's -mllvm flags, so use the real clang by full path.
RUN apt-get update -qq && apt-get install -y -qq clang > /dev/null && git clone -q --depth 1 -b v0.9.0 https://github.com/dzaima/CBQN /opt/cbqn && make -C /opt/cbqn CC=/usr/bin/clang > /tmp/cbqn-build.log 2>&1 && ln -s /opt/cbqn/BQN /usr/local/bin/bqn && bqn -p 1+1
# J 9.7 (jsoftware's Linux tarball has no version in its name: sha256 pinned). jconsole -> ijconsole (the JDK has a jconsole).
RUN curl -fsSL -o /tmp/j97.tgz https://www.jsoftware.com/download/j9.7/install/j9.7_linux64.tar.gz && echo "1c0cf450b45f563d7408a6c93563006782da8b422e0769781eb8731c465ffd06  /tmp/j97.tgz" | sha256sum -c && tar -xzf /tmp/j97.tgz -C /opt && rm /tmp/j97.tgz && ln -s /opt/j9.7/bin/jconsole /usr/local/bin/ijconsole
# ngn/k at pinned commit b9eeb91e0343ef6029f59d55dd98f82667b9bd59, built without -march=native (portable); clang as upstream uses (vendored git-archive fallback if codeberg.org is unreachable).
COPY vendor/ngnk-b9eeb91e.tar.gz /tmp/ngnk.tgz
RUN apt-get update -qq && apt-get install -y -qq clang > /dev/null && echo "3e0072b2a61e1247b58f424c1b818f94618aea87f5dc4bdbe94cbae766e9f7ac  /tmp/ngnk.tgz" | sha256sum -c && ( (git clone -q https://codeberg.org/ngn/k /opt/ngnk && cd /opt/ngnk && git checkout -q b9eeb91e0343ef6029f59d55dd98f82667b9bd59) || (rm -rf /opt/ngnk && mkdir -p /opt/ngnk && tar -xzf /tmp/ngnk.tgz -C /opt/ngnk) ) && rm -f /tmp/ngnk.tgz && make -C /opt/ngnk a N=k R=k O=-O3 L='-lm -ldl' CC=/usr/bin/clang > /tmp/ngnk-build.log 2>&1 && ln -s /opt/ngnk/k /usr/local/bin/ngnk
# Klong 20221212 (sha256 pinned).

# Q'Nial 7 core at a pinned commit (CMake 4 needs the policy floor; GCC 15 needs -std=gnu17 for its ctime call).
