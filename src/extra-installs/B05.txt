# B05: compile-to-JS languages. All six come from npm (pinned); PureScript's three base packages come from
# their GitHub release tags and Elm's two from package.elm-lang.org (both fetched once here, at build time).
RUN chronic apt-get -qq install -y npm && chronic apt-get clean
# one flat prefix, so wisp gets the commander it was written for (7.x; a global install pulls 14.x, which breaks it).
# Only the five compilers are linked into PATH: the prefix .bin also has "rc" and "which", which must not shadow the real ones.
RUN npm install --prefix /opt/b05 --no-fund --no-audit rescript@12.3.1 purescript@0.15.17 elm@0.19.1-6 imba@2.0.0-alpha.253 wisp@0.13.0 commander@7.2.0 pyret-npm@0.1.29 && for b in bsc purs elm imba wisp; do ln -s /opt/b05/node_modules/.bin/$b /usr/local/bin/$b; done
RUN for p in prelude:v6.0.2 effect:v4.0.0 console:v6.1.0; do mkdir -p /opt/purs/lib/${p%%:*} && curl -fsSL https://github.com/purescript/purescript-${p%%:*}/archive/refs/tags/${p#*:}.tar.gz | tar -xz --strip-components=1 -C /opt/purs/lib/${p%%:*} || exit 1; done && printf 'import {main} from "./out/QR/index.js"; main();\n' > /opt/purs/run.mjs && purs compile -o /opt/purs/out '/opt/purs/lib/*/src/**/*.purs'
RUN mkdir -p /opt/elm/src && cd /opt/elm && printf '{"type":"application","source-directories":["src"],"elm-version":"0.19.1","dependencies":{"direct":{"elm/core":"1.0.5","elm/json":"1.1.3"},"indirect":{}},"test-dependencies":{"direct":{},"indirect":{}}}\n' > elm.json && printf 'port module QR exposing (main)\nport o : String -> Cmd m\nmain : Program () () ()\nmain = Platform.worker {init = always ((), o "ok"), update = always (always ((), Cmd.none)), subscriptions = always Sub.none}\n' > src/QR.elm && elm make src/QR.elm --output=/dev/null
RUN printf '#!/bin/sh\n# pyretc IN.arr OUT.jarr: compile one Pyret program to a standalone node file\n# (the same compiler call the pyret CLI makes, without its background compile server)\nI=$(realpath "$1"); O=$(realpath -m "$2")\ncd %s && exec node --max-old-space-size=8192 --stack-size=8192 build/phaseA/pyret.jarr --build-runnable "$I" --outfile "$O" --builtin-js-dir src/js/trove/ --builtin-arr-dir src/arr/trove --require-config src/scripts/standalone-configA.json --standalone-file src/js/base/handalone.js --compiled-dir /opt/pyret-compiled -no-check-mode < /dev/null\n' /opt/b05/node_modules/pyret-npm/pyret-lang > /usr/local/bin/pyretc && chmod +x /usr/local/bin/pyretc && printf 'print("ok")' > /tmp/w.arr && pyretc /tmp/w.arr /tmp/w.jarr && NODE_PATH=/opt/b05/node_modules node /tmp/w.jarr
