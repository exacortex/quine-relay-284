# B20: Official Taxi, TryItOnline/rail, cipple (C Kipple interpreter), and Rockstar v2.0.15 (all pinned).
RUN mkdir -p /opt/taxi && curl --retry 5 -fsSL https://github.com/BigZaphod/Taxi/archive/61383585546668901e77287334ab454426e6f148.tar.gz | tar -xz -C /opt/taxi --strip-components=1 && g++ -O2 -std=gnu++14 -DNO_GREETINGS -include cstring -include cstdlib -include cstdint -include ctime -o /usr/local/bin/taxi /opt/taxi/source/taxi.cpp && rm -rf /opt/taxi && printf '%s\n' '"ok" is waiting at the Writer'\''s Depot.' 'Go to Writer'\''s Depot: west 1st left, 2nd right, 1st left, 2nd left.' 'Pickup a passenger going to the Post Office.' 'Go to the Post Office: north 1st right, 2nd right, 1st left.' 'Go to the Taxi Garage: north 1st right, 1st left, 1st right.' > /tmp/t.taxi && test "$(taxi /tmp/t.taxi)" = "ok" && rm -f /tmp/t.taxi
RUN mkdir -p /tmp/rail && curl --retry 5 -fsSL https://github.com/TryItOnline/rail/archive/04b7259fc19cceba2f06dfdc8431f683c81c80dd.tar.gz | tar -xz -C /tmp/rail --strip-components=1 && (cd /tmp/rail && bash compile.sh && install -m 755 /tmp/rail/rail /usr/local/bin/rail) && rm -rf /tmp/rail && printf "\$ 'main'\n \\\\\n  \\\\-[ok]o#\n" > /tmp/t.rail && test "$(rail /tmp/t.rail)" = "ok" && rm -f /tmp/t.rail
RUN curl --retry 5 -fsSL -o /tmp/cipple.c https://raw.githubusercontent.com/graue/esofiles/07dce2dbff3325ced047f080b4c9b9f433b3c72b/kipple/impl/cipple-0.2.1.c && gcc -O3 -w /tmp/cipple.c -o /usr/local/bin/cipple && rm -f /tmp/cipple.c && printf '107>o 111>o\n' > /tmp/t.kpl && test "$(cipple /tmp/t.kpl)" = "ok" && rm -f /tmp/t.kpl
RUN curl --retry 5 -fsSL -o /tmp/rock.tgz https://github.com/RockstarLang/rockstar/releases/download/v2.0.15/rockstar-v2.0.15-linux-x64.tar.gz && echo "cae38e48214f6512a593f8832e4905e11a4d2d218810a36641c1ac8f7eabfa74  /tmp/rock.tgz" | sha256sum -c && mkdir -p /tmp/rock && tar -xzf /tmp/rock.tgz -C /tmp/rock --strip-components=1 && install -m 755 /tmp/rock/rockstar /usr/local/bin/rockstar && rm -rf /tmp/rock /tmp/rock.tgz && printf 'shout "ok"\n' > /tmp/t.rock && test "$(rockstar /tmp/t.rock)" = "ok" && rm -f /tmp/t.rock
