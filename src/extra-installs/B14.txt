# B14: Dockerfile lines for installs with no apt package. Pinned (probe 2/3, Nov 14). Self-tests use no \n in echo.
# MoonScript 0.5.0 + Teal 0.24.8: luarocks rocks on Lua 5.4 (apt luarocks; moon/tl wrappers call lua5.4).
RUN (command -v luarocks > /dev/null || (apt-get update -qq && apt-get -qq install -y luarocks lua5.4 liblua5.4-dev && apt-get clean)) && luarocks --lua-version=5.4 install moonscript 0.5.0-1 > /tmp/moon.log 2>&1 && luarocks --lua-version=5.4 install tl 0.24.8-1 > /tmp/tl.log 2>&1 && echo 'io.write [==[moon ok]==]' > /tmp/t.moon && moon /tmp/t.moon && echo 'io.write([==[ tl ok]==])' > /tmp/t.tl && tl run /tmp/t.tl
# YueScript v0.34.3: standalone yue binary (bundles Lua 5.4), plain make (no rock for Lua 5.4).
RUN cd /opt && git clone -q --depth 1 -b v0.34.3 https://github.com/IppClub/YueScript && make -C YueScript release -j4 > /tmp/yue.log 2>&1 && ln -s /opt/YueScript/bin/release/yue /usr/local/bin/yue && echo 'io.write [==[yue ok]==]' > /tmp/t.yue && yue -e /tmp/t.yue
# TXR 302 (not packaged in Ubuntu 26.04): source snapshot, configure + make install (/usr/local).
RUN cd /opt && curl -fsSL --retry 6 --retry-delay 10 --retry-all-errors https://www.kylheku.com/cgit/txr/snapshot/txr-302.tar.bz2 | tar -xj && cd txr-302 && ./configure > /tmp/txr.log 2>&1 && make -j4 >> /tmp/txr.log 2>&1 && make install >> /tmp/txr.log 2>&1 && echo '(put-string "txr ok")' > /tmp/t.tl && txr /tmp/t.tl
# Arturo 0.10.0 "mini" build (the full build links libwebkit2gtk-4.0, gone from 26.04).
RUN mkdir -p /opt/arturo && cd /opt/arturo && curl -fsSL --retry 6 --retry-delay 10 --retry-all-errors https://github.com/arturo-lang/arturo/releases/download/v0.10.0/arturo-0.10.0-linux-amd64-mini.zip -o a.zip && unzip -q a.zip && rm a.zip && ln -s /opt/arturo/arturo /usr/local/bin/arturo && echo 'prints {:arturo ok:}' > /tmp/t.art && arturo /tmp/t.art
# Ring v1.27: only the core interpreter (language/src, ~45 C files; ringw.c is the Windows GUI entry and is left out).
RUN cd /opt && git clone -q --depth 1 --filter=blob:none --sparse -b v1.27 https://github.com/ring-lang/ring && cd ring && git sparse-checkout set language > /dev/null && cd language/src && gcc -O2 $(ls *.c | grep -v ringw) -I ../include -lm -ldl -o /usr/local/bin/ring > /tmp/ring.log 2>&1 && echo 'see "ring ok"' > /tmp/t.ring && ring /tmp/t.ring
